\ExplSyntaxOn
\tl_new:N \l__wm_frame_options_tl

\cs_new_protected:Npn \WMFramePrepareOptions #1
  {
    \wmframeRenderfalse
    \wmframeGridfalse
    \wmframeCleanfalse
    \tl_clear:N \l__wm_frame_options_tl
    \xdef\WMFrameVSpace{\noexpand\vspace{0cm}}
    \gdef\WMFrameAlign{}
    \clist_map_inline:nn {#1}
      {
        \tl_set:Nn \l_tmpa_tl {##1}
        \tl_trim_spaces:N \l_tmpa_tl
        \str_case:VnF \l_tmpa_tl
          {
            {render}
              {
                \wmframeRendertrue
              }
            {grid}
              {
                \wmframeGridtrue
              }
            {clean}
              {
                \wmframeCleantrue
              }
            {t}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{1cm}}
                \tl_if_empty:NTF \l__wm_frame_options_tl
                  {\tl_put_right:Nn \l__wm_frame_options_tl {t}}
                  {\tl_put_right:Nn \l__wm_frame_options_tl {,t}}
              }
            {T}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{1cm}}
                \tl_if_empty:NTF \l__wm_frame_options_tl
                  {\tl_put_right:Nn \l__wm_frame_options_tl {t}}
                  {\tl_put_right:Nn \l__wm_frame_options_tl {,t}}
              }
            {c}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{0cm}}
                \gdef\WMFrameAlign{}
              }
            {C}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{0cm}}
                \gdef\WMFrameAlign{}
              }
            {ct}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{0cm}}
                \gdef\WMFrameAlign{\centering}
              }
            {CT}
              {
                \xdef\WMFrameVSpace{\noexpand\vspace{0cm}}
                \gdef\WMFrameAlign{\centering}
              }
          }
          {
            \tl_if_blank:VF \l_tmpa_tl
              {
                \tl_if_empty:NTF \l__wm_frame_options_tl
                  {
                    \exp_args:NV \tl_set:Nn \l__wm_frame_options_tl \l_tmpa_tl
                  }
                  {
                    \tl_put_right:Nn \l__wm_frame_options_tl {,}
                    \tl_put_right:NV \l__wm_frame_options_tl \l_tmpa_tl
                  }
              }
          }
      }
    \xdef\WMFrameOptions{\tl_use:N \l__wm_frame_options_tl}
  }
\ExplSyntaxOff

\newcommand{\WMDrawGrid}{%
    \ifwmframeGrid
        \begin{tikzpicture}[remember picture,overlay]
            \begin{scope}[shift={(current page.south west)}]
                \draw[step=0.5cm,black!12,line width=0.1pt]
                    (0,0) grid (\paperwidth,\paperheight);
                \draw[step=1cm,black!38,line width=0.2pt]
                    (0,0) grid (\paperwidth,\paperheight);
            \end{scope}
            \draw[SectionTitleColor!70,dashed,line width=0.4pt]
                (current page.south) -- (current page.north);
            \draw[SectionTitleColor!70,dashed,line width=0.4pt]
                (current page.west) -- (current page.east);
            \draw[black!55,line width=0.35pt]
                (current page.south west) -- (current page.south east);
            \draw[black!55,line width=0.35pt]
                (current page.south west) -- (current page.north west);

            \pgfmathtruncatemacro{\WMGridLastX}{\paperwidth/1cm-1}
            \pgfmathtruncatemacro{\WMGridLastY}{\paperheight/1cm-1}
            \foreach \x in {1,...,\WMGridLastX}{%
                \node[
                    anchor=south,
                    inner sep=0.5pt,
                    fill=white,
                    fill opacity=0.75,
                    text opacity=1,
                    font=\fontsize{5}{5}\selectfont,
                    text=black!85
                ] at ([xshift=\x cm,yshift=0.05cm]current page.south west) {\x};
                \node[
                    anchor=north,
                    inner sep=0.5pt,
                    fill=white,
                    fill opacity=0.75,
                    text opacity=1,
                    font=\fontsize{5}{5}\selectfont,
                    text=black!85
                ] at ([xshift=\x cm,yshift=-0.05cm]current page.north west) {\x};
            }
            \foreach \y in {1,...,\WMGridLastY}{%
                \node[
                    anchor=west,
                    inner sep=0.5pt,
                    fill=white,
                    fill opacity=0.75,
                    text opacity=1,
                    font=\fontsize{5}{5}\selectfont,
                    text=black!85
                ] at ([xshift=0.05cm,yshift=\y cm]current page.south west) {\y};

                    anchor=east,
                    inner sep=0.5pt,
                    fill=white,
                    fill opacity=0.75,
                    text opacity=1,
                    font=\fontsize{5}{5}\selectfont,
                    text=black!85

                anchor=south west,

                fill opacity=0.75,

                font=\fontsize{5}{5}\selectfont,
                text=black!85
            ] at ([xshift=0.05cm,yshift=0.05cm]current page.south west) {0};
        \end{tikzpicture}%
    \fi
}

\NewDocumentCommand{\Frame}{O{} +m}{%
    \WMFramePrepareOptions{#1}
    \begingroup
    \ifdefempty{\subsecname}{%
        \renewcommand{\WMFrameTitleYShift}{-0.48cm}%
    }{%
        \renewcommand{\WMFrameTitleYShift}{-0.25cm}%
    }%
    \ifwmframeRender
        \setkeys{Gin}{draft=false}%
    \fi
    \ifwmframeClean
        \setbeamertemplate{navigation symbols}{}%
        \begin{frame}[plain]
            \centering
            \begin{minipage}{0.98\textwidth}
                \MyConfig
                \WMFrameAlign#2
            \end{minipage}%
            \WMDrawGrid
        \end{frame}
    \else
        \ifdefempty{\WMFrameOptions}{%
            \begin{frame}{\secname}
        }{%
            \begin{frame}[\WMFrameOptions]{\secname}
        }%
            \framesubtitle{\subsecname}
            \WMFrameVSpace
            \centering
            \begin{minipage}{0.98\textwidth}
                \MyConfig
                \WMFrameAlign#2
            \end{minipage}%
            \WMDrawGrid
        \end{frame}
    \fi
    \endgroup
    \wmframeCleanfalse
    \wmframeGridfalse
}

\ExplSyntaxOn
\seq_new:N \g_wm_title_authors_seq
\seq_new:N \g_wm_title_affiliations_seq

\cs_new:Npn \wm_title_authors:
  {
    \seq_map_indexed_inline:Nn \g_wm_title_authors_seq
      {
        \int_compare:nNnT {##1} > {1}
          {
            \int_compare:nNnTF {##1} = {\seq_count:N \g_wm_title_authors_seq}
              {
                \int_compare:nNnTF {\seq_count:N \g_wm_title_authors_seq} = {2}
                  {~and~}
                  {,~and~}
              }
              {,~}
          }
        ##2
      }
  }

  {
    \seq_map_indexed_inline:Nn \g_wm_title_affiliations_seq
      {
        \int_compare:nNnT {##1} > {1} {\\}
        ##2
      }
  }
