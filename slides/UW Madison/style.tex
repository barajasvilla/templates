\ExplSyntaxOn
\clist_new:N \l__wm_frame_options_clist
\bool_new:N \l__wm_frame_top_bool

\cs_new_protected:Npn \WMFramePrepareOptions #1
  {
    \wmframeRenderfalse
    \wmframeGridfalse
    \wmframeCleanfalse
    \clist_clear:N \l__wm_frame_options_clist
    \bool_set_true:N \l__wm_frame_top_bool
    \def\WMFrameVSpace{\vspace{1cm}}
    \def\WMFrameAlign{}
    \clist_map_inline:nn {#1}
      {
        \tl_set:Nn \l_tmpa_tl {##1}
        \tl_trim_spaces:N \l_tmpa_tl
        \tl_set:Nx \l_tmpb_tl {\str_lowercase:f {\l_tmpa_tl}}
        \str_case:VnF \l_tmpb_tl
          {
            {render}
              {\wmframeRendertrue}
            {r}
              {\wmframeRendertrue}
            {grid}
              {\wmframeGridtrue}
            {g}
              {\wmframeGridtrue}
            {clean}
              {\wmframeCleantrue}
            {t}
              {
                \bool_set_true:N \l__wm_frame_top_bool
                \def\WMFrameVSpace{\vspace{1cm}}
                \def\WMFrameAlign{}
              }
            {c}
              {
                \bool_set_false:N \l__wm_frame_top_bool
                \def\WMFrameVSpace{}
                \def\WMFrameAlign{}
              }
            {cc}
              {
                \bool_set_false:N \l__wm_frame_top_bool
                \def\WMFrameVSpace{}
                \def\WMFrameAlign{\centering}
              }
          }
          {
            \tl_if_blank:VF \l_tmpa_tl
              {\clist_put_right:NV \l__wm_frame_options_clist \l_tmpa_tl}
          }
      }
    \bool_if:NT \l__wm_frame_top_bool
      {\clist_put_right:Nn \l__wm_frame_options_clist {t}}
    \edef\WMFrameOptions{\clist_use:Nn \l__wm_frame_options_clist {,}}
  }
\ExplSyntaxOff

% Interpret bare numeric lengths as centimetres while preserving explicit
% units, dimension expressions, and symbolic values such as `c`.
\ExplSyntaxOn
\regex_const:Nn \c__wm_numeric_length_regex
  { \A (?: \+ | - )? (?: \d+ (?: \. \d* )? | \. \d+ ) \Z }
\cs_new_protected:Npn \wm_normalize_length:nN #1#2
  {
    \tl_set:Nx \l_tmpa_tl {#1}
    \tl_trim_spaces:N \l_tmpa_tl
    \regex_match:NVTF
      \c__wm_numeric_length_regex
      \l_tmpa_tl
      { \tl_put_right:Nn \l_tmpa_tl { cm } }
      { }
    \cs_set:Npx #2 { \exp_not:V \l_tmpa_tl }
  }
\NewDocumentCommand{\WMNormalizeLength}{m m}
  { \wm_normalize_length:nN {#1} #2 }
\ExplSyntaxOff

    \pgfkeys{/Note/.cd,defaults,#1}%
    \edef\WMNoteX{\pgfkeysvalueof{/Note/x}}%
    \edef\WMNoteY{\pgfkeysvalueof{/Note/y}}%
    \edef\WMNoteColor{\pgfkeysvalueof{/Note/color}}%
    \edef\WMNoteWidth{\pgfkeysvalueof{/Note/width}}%
    \WMNormalizeLength{\WMNoteX}{\WMNoteX}%
    \WMNormalizeLength{\WMNoteY}{\WMNoteY}%
    \WMNormalizeLength{\WMNoteWidth}{\WMNoteWidth}%
    \def\WMNoteInnerXSep{8pt}%
    \def\WMNoteLineWidth{0.9pt}%
    \ifdefstring{\WMNoteX}{c}{\def\WMNoteX{0.5\paperwidth}}{}%
    \ifdefstring{\WMNoteY}{c}{\def\WMNoteY{0.5\paperheight}}{}%
    \begin{tikzpicture}[remember picture,overlay]
        \node[
            name=wmnotebox,
            anchor=\pgfkeysvalueof{/Note/anchor},
            fill=\WMNoteColor!5!white,
            draw=\WMNoteColor!70!black,
            dash pattern=on 4pt off 3pt,
            line width=\WMNoteLineWidth,
            rounded corners=3pt,
            inner xsep=\WMNoteInnerXSep,
            inner ysep=6pt,
            outer sep=0pt,
            xshift=\WMNoteX,
            yshift=\WMNoteY
        ] at (\pgfkeysvalueof{/Note/at}) {%
            \begin{minipage}[t]{\dimexpr
                \WMNoteWidth
                -\WMNoteInnerXSep
                -\WMNoteInnerXSep
                -\WMNoteLineWidth
                \relax}
                \vspace{0pt}%
                \MyConfig
                \raggedright
                #2
            \end{minipage}%
        };
        \ifwmframeGrid
            \draw[blue!75!black,dashed,line width=0.8pt]
                (wmnotebox.south west) rectangle (wmnotebox.north east);
            \draw[blue!75!black,line width=1.2pt]
                ([yshift=-3pt]wmnotebox.north) -- ([yshift=3pt]wmnotebox.north)
                ([yshift=-3pt]wmnotebox.south) -- ([yshift=3pt]wmnotebox.south)
                ([xshift=-3pt]wmnotebox.west) -- ([xshift=3pt]wmnotebox.west)
                ([xshift=-3pt]wmnotebox.east) -- ([xshift=3pt]wmnotebox.east);
            \fill[blue!75!black]
                ([xshift=\WMNoteX,yshift=\WMNoteY]\pgfkeysvalueof{/Note/at})
                circle[radius=2pt];
        \fi
    \end{tikzpicture}%
